## Usage

Render each variant you show on the server, then pass them in:

```tsx
import { listVariants, render } from "@copybook/core";

import { CodeVariants } from "@/components/copybook/code-variants";
import { tryIt } from "@/content/ai-gateway";

export const TryIt = ({ endpoint, model }) => (
  <CodeVariants
    items={listVariants(tryIt)
      .filter((variant) => variant.when.scenario === "chat")
      .map((variant) => ({
        key: variant.key,
        label: variant.label,
        rendered: render(tryIt, { endpoint, model }, { variant: variant.key }),
      }))}
  />
);
```

## Props

| Prop | Type | Description |
| --- | --- | --- |
| `items` | `{ key, label, rendered }[]` | One tab each, in order. |
| `defaultKey` | `string` | The tab that starts selected. |
| `onCopy` | `(event: CopyEvent) => void` | Called with `{ id, variant, text }` after a copy. |
